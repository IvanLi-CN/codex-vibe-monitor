describe("UpstreamAccountCreatePage api key", () => {
  it("hides email and mother fields for api key creation and never submits them", async () => {
    const createApiKeyAccount = vi.fn().mockResolvedValue({
      id: 42,
      kind: "api_key_codex",
      provider: "codex",
      displayName: "Gateway Key",
      groupName: null,
      isMother: false,
      status: "active",
      enabled: true,
      history: [],
    });
    mockUpstreamAccounts({ createApiKeyAccount });
    render("/account-pool/upstream-accounts/new?mode=apiKey");

    expect(document.body.querySelector('input[name="apiKeyEmail"]')).toBeNull();
    expect(document.body.textContent || "").not.toContain("Set as mother");
    setInputValue('input[name="apiKeyDisplayName"]', "Gateway Key");
    setInputValue('input[name="apiKeyValue"]', "sk-gateway");
    clickButton(/Create API Key account/i);
    await flushAsync();

    const payload = createApiKeyAccount.mock.calls[0]?.[0];
    expect(payload).toMatchObject({
      displayName: "Gateway Key",
      apiKey: "sk-gateway",
      boundProxyKeys: ["__direct__"],
    });
    expect(payload).not.toHaveProperty("email");
    expect(payload).not.toHaveProperty("isMother");
  });

  it("keeps the direct transit binding when the final node is toggled", async () => {
    const createApiKeyAccount = vi.fn().mockResolvedValue({
      id: 42,
      kind: "api_key_codex",
      provider: "codex",
      displayName: "Gateway Key",
      groupName: null,
      isMother: false,
      status: "active",
      enabled: true,
      history: [],
    });
    mockUpstreamAccounts({ createApiKeyAccount });
    render("/account-pool/upstream-accounts/new?mode=apiKey");

    setInputValue('input[name="apiKeyDisplayName"]', "Gateway Key");
    setInputValue('input[name="apiKeyValue"]', "sk-gateway");
    clickButton(/^Direct/);
    clickButton(/Create API Key account/i);
    await flushAsync();

    expect(createApiKeyAccount).toHaveBeenCalledWith(
      expect.objectContaining({ boundProxyKeys: ["__direct__"] }),
    );
  });

  it("submits upstreamBaseUrl for API key accounts", async () => {
    const createApiKeyAccount = vi.fn().mockResolvedValue({
      id: 42,
      kind: "api_key_codex",
      provider: "codex",
      displayName: "Gateway Key",
      groupName: null,
      isMother: false,
      status: "active",
      enabled: true,
      history: [],
    });
    mockUpstreamAccounts({ createApiKeyAccount });
    render("/account-pool/upstream-accounts/new?mode=apiKey");

    setInputValue('input[name="apiKeyDisplayName"]', "Gateway Key");
    setInputValue('input[name="apiKeyValue"]', "sk-gateway");
    setInputValue(
      'input[name="apiKeyUpstreamBaseUrl"]',
      "https://proxy.example.com/gateway",
    );

    clickButton(/Create API Key account/i);
    await flushAsync();

    expect(createApiKeyAccount).toHaveBeenCalledWith(
      expect.objectContaining({
        displayName: "Gateway Key",
        apiKey: "sk-gateway",
        upstreamBaseUrl: "https://proxy.example.com/gateway",
      }),
    );
  });

  it("blocks api key creation when upstreamBaseUrl is not a valid absolute URL", () => {
    const createApiKeyAccount = vi.fn().mockResolvedValue({
      id: 42,
      kind: "api_key_codex",
      provider: "codex",
      displayName: "Gateway Key",
      groupName: null,
      isMother: false,
      status: "active",
      enabled: true,
      history: [],
    });
    mockUpstreamAccounts({ createApiKeyAccount });
    render("/account-pool/upstream-accounts/new?mode=apiKey");

    setInputValue('input[name="apiKeyDisplayName"]', "Gateway Key");
    setInputValue('input[name="apiKeyValue"]', "sk-gateway");
    setInputValue(
      'input[name="apiKeyUpstreamBaseUrl"]',
      "proxy.example.com/gateway",
    );

    expect(document.body.textContent).toContain("Use an absolute http(s) URL");
    expect(findButton(/Create API Key account/i)?.disabled).toBe(true);
    expect(createApiKeyAccount).not.toHaveBeenCalled();
  });

  it("does not render group controls or submit group metadata", async () => {
    const createApiKeyAccount = vi.fn().mockResolvedValue({
      id: 42,
      kind: "api_key_codex",
      provider: "codex",
      displayName: "Gateway Key",
      groupName: null,
      isMother: false,
      status: "active",
      enabled: true,
      history: [],
    });
    mockUpstreamAccounts({ createApiKeyAccount });
    render("/account-pool/upstream-accounts/new?mode=apiKey");

    setInputValue('input[name="apiKeyDisplayName"]', "Gateway Key");
    setInputValue('input[name="apiKeyValue"]', "sk-gateway");
    expect(document.body.querySelector('input[name="apiKeyGroupName"]')).toBeNull();

    clickButton(/Create API Key account/i);
    await flushAsync();

    expect(createApiKeyAccount).toHaveBeenCalledWith(
      expect.objectContaining({
        apiKey: "sk-gateway",
      }),
    );
    expect(window.localStorage.setItem).not.toHaveBeenCalledWith(
      UPSTREAM_ACCOUNT_CREATE_API_KEY_LAST_GROUP_STORAGE_KEY,
      expect.any(String),
    );
  });

  it("keeps API Key creation failures on the transit form", async () => {
    const createApiKeyAccount = vi
      .fn()
      .mockRejectedValue(new Error("Request failed: 500 upstream rejected"));
    mockUpstreamAccounts({ createApiKeyAccount });
    render("/account-pool/upstream-accounts/new?mode=apiKey");

    setInputValue('input[name="apiKeyDisplayName"]', "Gateway Key");
    setInputValue('input[name="apiKeyValue"]', "sk-gateway");
    clickButton(/Create API Key account/i);
    await flushAsync();

    expect(createApiKeyAccount).toHaveBeenCalled();
    expect(document.body.textContent || "").toContain("upstream rejected");
    expect(window.localStorage.setItem).not.toHaveBeenCalledWith(
      UPSTREAM_ACCOUNT_CREATE_API_KEY_LAST_GROUP_STORAGE_KEY,
      expect.any(String),
    );
  });
});
