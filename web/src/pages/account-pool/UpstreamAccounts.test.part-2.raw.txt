    healthStatus: [],
    tagIds: [],
    groupFilter: {
      mode: "search",

    },
  });

    pathname: "/account-pool/upstream-accounts",
    state: {
      presetGroupFilter: {
        mode: "exact",

      },
    },
  });

  await flushAsync();
  await flushAsync();

  expect(hookMocks.useUpstreamAccounts.mock.calls[0]?.[0]).toEqual({

    tagIds: undefined,
  });

    includeAll: true,

    },
