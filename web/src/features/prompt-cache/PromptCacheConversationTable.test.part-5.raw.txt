function applyPolicyBindingPatch(
  current: PromptCacheConversationBindingResponse,
  payload: UpdatePromptCacheConversationBindingPayload,
): PromptCacheConversationBindingResponse {
  const timeoutPatch = payload.timeouts ?? {};
  const defaults = {
    responsesFirstByteTimeoutSecs: 120,
    compactFirstByteTimeoutSecs: 300,
    responsesStreamTimeoutSecs: 300,
    compactStreamTimeoutSecs: 300,
  };
  const timeoutValue = (key: keyof typeof defaults) =>
    key in timeoutPatch ? (timeoutPatch[key] ?? defaults[key]) : current.timeouts[key];
  const forwardProxyKeys = Array.isArray(payload.forwardProxyKeys)
    ? payload.forwardProxyKeys
    : "forwardProxyKey" in payload && payload.forwardProxyKey
      ? [payload.forwardProxyKey]
      : (current.forwardProxyKeys ?? []);
  return {
    ...current,
    bindingKind: payload.bindingKind,
    allowSwitchUpstream: payload.allowSwitchUpstream ?? current.allowSwitchUpstream,
    fastModeRewriteMode: payload.fastModeRewriteMode ?? current.fastModeRewriteMode,
    imageToolRewriteMode: payload.imageToolRewriteMode ?? current.imageToolRewriteMode,
    availableModels: payload.availableModels ?? current.availableModels,
    forwardProxyKey: forwardProxyKeys[0] ?? null,
    forwardProxyKeys,
    timeouts: {
      responsesFirstByteTimeoutSecs: timeoutValue("responsesFirstByteTimeoutSecs"),
      compactFirstByteTimeoutSecs: timeoutValue("compactFirstByteTimeoutSecs"),
      responsesStreamTimeoutSecs: timeoutValue("responsesStreamTimeoutSecs"),
      compactStreamTimeoutSecs: timeoutValue("compactStreamTimeoutSecs"),
    },
    updatedAt: "2026-03-02T12:01:00Z",
  };
}
